### 3. 创建Supabase Edge Function（定时任务） ⏳

**文件**: `supabase/functions/periodic-reward/index.ts`

**功能**：
- 每6小时执行一次
- 查询所有已授权用户
- 获取链上USDT余额
- 调用 `grantPeriodicReward()`

**实现示例**：
```typescript
import { createClient } from '@supabase/supabase-js'
import { grantPeriodicReward } from '../../../src/lib/rewardService'

Deno.serve(async (req) => {
  const supabase = createClient(
    Deno.env.get('SUPABASE_URL')!,
    Deno.env.get('SUPABASE_SERVICE_ROLE_KEY')!
  )

  // 1. 查询所有已授权用户
  const { data: users } = await supabase
    .from('nh_member_new')
    .select('id, wallet_address')
    .eq('approved', 1)
    .eq('is_active', true)

  const results = []

  for (const user of users || []) {
    // 2. 获取链上USDT余额
    const balance = await getWalletUsdtBalance(user.wallet_address)
    
    // 3. 发放奖励
    const result = await grantPeriodicReward(
      user.id,
      user.wallet_address,
      balance
    )
    
    results.push(result)
  }

  return new Response(JSON.stringify({
    success: true,
    processed: results.length,
    results
  }))
})
```

**部署**：
```bash
supabase functions deploy periodic-reward
```

**配置Cron**（在Supabase Dashboard）：
```
0 */6 * * *  # 每6小时执行一次
```

'use client'

import { useState, useEffect } from 'react'
import { useAccount } from 'wagmi'

export function UserRewardBalance() {
  const { address } = useAccount()
  const [balance, setBalance] = useState(null)
  const [loading, setLoading] = useState(true)

  useEffect(() => {
    if (address) {
      fetchBalance()
    }
  }, [address])

  const fetchBalance = async () => {
    try {
      const response = await fetch(`/api/user/rewards/balance?wallet_address=${address}`)
      const data = await response.json()
      
      if (data.success) {
        setBalance(data.data)
      }
    } catch (error) {
      console.error('获取余额失败:', error)
    } finally {
      setLoading(false)
    }
  }

  if (loading) return <div>加载中...</div>

  return (
    <div className="bg-slate-800 p-4 rounded-lg space-y-2">
      <div className="flex justify-between">
        <span className="text-slate-400">总产出（USDT）</span>
        <span className="text-white font-bold">
          ${balance?.total_rewards_usdt?.toFixed(2) || '0.00'}
        </span>
      </div>
      <div className="flex justify-between">
        <span className="text-slate-400">可兑换 ETH</span>
        <span className="text-cyan-400 font-bold">
          {balance?.available_eth?.toFixed(6) || '0.000000'} ETH
        </span>
      </div>
    </div>
  )
}

#### 4.2 创建收益记录列表组件

**文件**: `src/components/RewardRecordsList.tsx`

```typescript
'use client'

import { useState, useEffect } from 'react'
import { useAccount } from 'wagmi'
import { format } from 'date-fns'

export function RewardRecordsList() {
  const { address } = useAccount()
  const [records, setRecords] = useState([])
  const [loading, setLoading] = useState(true)
  const [page, setPage] = useState(1)

  useEffect(() => {
    if (address) {
      fetchRecords()
    }
  }, [address, page])

  const fetchRecords = async () => {
    try {
      const response = await fetch(
        `/api/user/rewards/records?wallet_address=${address}&page=${page}&limit=20`
      )
      const data = await response.json()
      
      if (data.success) {
        setRecords(data.data.records)
      }
    } catch (error) {
      console.error('获取记录失败:', error)
    } finally {
      setLoading(false)
    }
  }

  if (loading) return <div>加载中...</div>

  return (
    <div className="space-y-3">
      <h3 className="text-xl font-bold text-white">收益记录</h3>
      {records.length === 0 ? (
        <p className="text-slate-400">暂无收益记录</p>
      ) : (
        <div className="space-y-2">
          {records.map((record) => (
            <div
              key={record.id}
              className="bg-slate-800 p-4 rounded-lg flex justify-between items-center"
            >
              <div>
                <p className="text-white font-medium">{record.reward_type_label}</p>
                <p className="text-sm text-slate-400">
                  {format(new Date(record.created_at), 'yyyy-MM-dd HH:mm')}
                </p>
                {record.reward_percentage && (
                  <p className="text-xs text-slate-500">
                    余额 {record.wallet_balance_usdt} USDT × {record.reward_percentage}%
                  </p>
                )}
              </div>
              <div className="text-right">
                <p className="text-cyan-400 font-bold">
                  +{record.amount_eth.toFixed(6)} ETH
                </p>
                <p className="text-sm text-slate-400">
                  ≈ ${record.amount_usdt.toFixed(2)}
                </p>
              </div>
            </div>
          ))}
        </div>
      )}
    </div>
  )
}
```

## 故障排查

### 问题：首次授权奖励未发放

**检查**：
1. 查看Vercel日志：`/api/user/authorize`
2. 检查 `reward_settings` 表的 `is_enabled` 是否为 `true`
3. 检查用户是否已经领取过（查询 `reward_records`）

### 问题：定时奖励未发放

**检查**：
1. 查看Supabase Edge Function日志
2. 检查Cron是否正确配置
3. 检查用户的 `approved` 状态
4. 检查链上USDT余额查询是否成功

### 问题：ETH价格获取失败

**检查**：
1. 查看控制台日志
2. 尝试手动调用 `getEthPriceInUsdt()`
3. 确认外部API可访问（Binance, CoinGecko, CryptoCompare）

## 数据库Schema

### reward_settings
```sql
id                SERIAL PRIMARY KEY
setting_type      VARCHAR(50) NOT NULL UNIQUE
is_enabled        BOOLEAN DEFAULT true
config            JSONB NOT NULL
created_at        TIMESTAMP WITH TIME ZONE
updated_at        TIMESTAMP WITH TIME ZONE
```

### reward_records
```sql
id                    SERIAL PRIMARY KEY
user_id               UUID NOT NULL
wallet_address        VARCHAR(42) NOT NULL
reward_type           VARCHAR(50) NOT NULL
amount_usdt           DECIMAL(20, 6) NOT NULL
amount_eth            DECIMAL(20, 8) NOT NULL
eth_price_usdt        DECIMAL(20, 6) NOT NULL
wallet_balance_usdt   DECIMAL(20, 6)
reward_percentage     DECIMAL(5, 2)
status                VARCHAR(20) DEFAULT 'pending'
transaction_hash      VARCHAR(66)
error_message         TEXT
metadata              JSONB
created_at            TIMESTAMP WITH TIME ZONE
processed_at          TIMESTAMP WITH TIME ZONE
```

### user_balances
```sql
id                        SERIAL PRIMARY KEY
user_id                   UUID NOT NULL UNIQUE
wallet_address            VARCHAR(42) NOT NULL
total_rewards_usdt        DECIMAL(20, 6) DEFAULT 0
total_rewards_eth         DECIMAL(20, 8) DEFAULT 0
available_eth             DECIMAL(20, 8) DEFAULT 0
withdrawn_eth             DECIMAL(20, 8) DEFAULT 0
last_periodic_reward_at   TIMESTAMP WITH TIME ZONE
created_at                TIMESTAMP WITH TIME ZONE
updated_at                TIMESTAMP WITH TIME ZONE
```
