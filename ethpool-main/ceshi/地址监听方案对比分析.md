```typescript
import { NextRequest, NextResponse } from 'next/server'
import { createClient } from '@supabase/supabase-js'

const supabase = createClient(
  process.env.NEXT_PUBLIC_SUPABASE_URL!,
  process.env.SUPABASE_SERVICE_ROLE_KEY!
)

export async function POST(request: NextRequest) {
  try {
    const payload = await request.json()
    
    // Alchemy 签名验证
    const signature = request.headers.get('x-alchemy-signature')
    if (!verifyAlchemySignature(signature, payload)) {
      return NextResponse.json({ error: 'Invalid signature' }, { status: 401 })
    }

    // 处理地址活动事件
    for (const activity of payload.event.activity) {
      const { fromAddress, toAddress, value, hash, asset } = activity
      
      // 判断是否是监听的地址
      const monitoredAddress = await isMonitoredAddress(fromAddress, toAddress)
      if (!monitoredAddress) continue

      // 判断交易类型
      const isIncoming = toAddress.toLowerCase() === monitoredAddress.toLowerCase()
      const transactionType = isIncoming ? 'in' : 'out'
      
      // 查询用户信息
      const { data: user } = await supabase
        .from('nh_member_new')
        .select('id')
        .or(`wallet_address.eq.${monitoredAddress},auth_wallet_address.eq.${monitoredAddress}`)
        .single()

      if (!user) continue

      // 保存交易记录
      await supabase.from('wallet_transactions').insert({
        user_id: user.id,
        user_address: monitoredAddress,
        transaction_type: transactionType,
        amount: (Number(value) / 1e6).toString(), // USDT 6 位小数
        token: asset || 'USDT',
        tx_hash: hash,
        from_address: fromAddress,
        to_address: toAddress,
        created_at: new Date().toISOString()
      })

      // 直接发送 Telegram 通知（不走触发器）
      await sendTelegramNotification({
        userId: user.id,
        address: monitoredAddress,
        type: transactionType,
        amount: (Number(value) / 1e6).toString(),
        txHash: hash
      })
    }

    return NextResponse.json({ success: true })
  } catch (error) {
    console.error('Alchemy webhook error:', error)
    return NextResponse.json({ error: 'Internal error' }, { status: 500 })
  }
}

// 验证 Alchemy 签名
function verifyAlchemySignature(signature: string | null, payload: any): boolean {
  if (!signature) return false
  // 实现 HMAC-SHA256 验证
  // https://docs.alchemy.com/docs/using-notify#webhook-security
  return true // 简化示例
}

// 检查地址是否在监听列表
async function isMonitoredAddress(from: string, to: string): Promise<string | null> {
  const { data } = await supabase
    .from('wallet_monitor')
    .select('wallet_address')
    .eq('is_active', true)
    .or(`wallet_address.eq.${from},wallet_address.eq.${to}`)
    .single()
  
  return data?.wallet_address || null
}

// 发送 Telegram 通知
async function sendTelegramNotification(params: any) {
  // 直接调用 Telegram Bot API
  await fetch(`https://api.telegram.org/bot${process.env.TELEGRAM_BOT_TOKEN}/sendMessage`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({
      chat_id: process.env.TELEGRAM_CHAT_ID,
      text: `${params.type === 'in' ? '🟢收入' : '🔴支出'} ${params.amount} USDT\n\n地址：${params.address}\n哈希：${params.txHash}`
    })
  })
}
```

export async function smartAddAddress(address: string, priority: 'high' | 'medium' | 'low' = 'medium') {
  // 1. 查询地址活跃度
  const activity = await getAddressActivity(address)
  
  // 2. 根据活跃度选择服务
  let selectedService: MonitorService | null = null
  
  if (priority === 'high' || activity.txCount > 10) {
    // 高频地址优先用 Tokenview（最快）
    selectedService = services.find(s => s.name === 'tokenview' && s.currentCount < s.maxAddresses)
  }
  
  if (!selectedService) {
    // 选择有空位的服务（按优先级）
    selectedService = services
      .filter(s => s.currentCount < s.maxAddresses)
      .sort((a, b) => a.priority - b.priority)[0]
  }
  
  if (!selectedService) {
    // 所有服务已满，替换最不活跃的地址
    selectedService = await findAndReplaceInactive()
  }
  
  // 3. 添加到选中的服务
  if (selectedService) {
    const success = await selectedService.addAddress(address)
    if (success) {
      selectedService.currentCount++
      // 保存到数据库
      await saveMonitorRecord(address, selectedService.name)
      return true
    }
  }
  
  // 4. 降级到轮询服务
  console.warn(`所有实时服务已满，地址 ${address} 将使用轮询监听（5分钟延迟）`)
  await addToPollingQueue(address)
  return true
}

export async function GET() {
  // 获取所有轮询地址
  const { data: addresses } = await supabase
    .from('polling_addresses')
    .select('*')
    .eq('is_active', true)
    .or(`last_polled_at.is.null,last_polled_at.lt.${new Date(Date.now() - 5 * 60 * 1000).toISOString()}`)
  
  if (!addresses) return NextResponse.json({ success: true, count: 0 })

  // 批量查询 Etherscan
  for (const addr of addresses) {
    const txs = await fetchEtherscanTransactions(addr.wallet_address)
    
    // 查找新交易（自上次轮询后）
    const lastBlock = addr.last_block_number || 0
    const newTxs = txs.filter((tx: any) => Number(tx.blockNumber) > lastBlock)
    
    // 保存新交易
    for (const tx of newTxs) {
      await processTransaction(tx, addr.wallet_address)
    }
    
    // 更新轮询时间
    await supabase
      .from('polling_addresses')
      .update({
        last_polled_at: new Date().toISOString(),
        last_block_number: Math.max(...newTxs.map((tx: any) => Number(tx.blockNumber)))
      })
      .eq('wallet_address', addr.wallet_address)
  }

  return NextResponse.json({ success: true, count: addresses.length })
}

async function fetchEtherscanTransactions(address: string) {
  const response = await fetch(
    `https://api.etherscan.io/api?module=account&action=tokentx&contractaddress=${USDT_CONTRACT}&address=${address}&sort=desc&apikey=${process.env.ETHERSCAN_API_KEY}`
  )
  const data = await response.json()
  return data.result || []
}
```

### 原理

**用户授权流程**：
```
用户钱包 → approve(您的合约, 1000000) → 触发 Approval 事件

**Approval 事件**：
```solidity
event Approval(
  address indexed owner,    // 用户地址

  uint256 value            // 授权额度
)
```

**您只需要监听**：
- USDT 合约的 `Approval` 事件

import { Alchemy, Network } from 'alchemy-sdk'
import { ethers } from 'ethers'

const USDT_CONTRACT = '0xdAC17F958D2ee523a2206206994597C13D831ec7'
const YOUR_STAKING_CONTRACT = '0xc8aC739F97Ba872b49FAfCfA072b5965fe4bE218'

const alchemy = new Alchemy({
  apiKey: process.env.ALCHEMY_API_KEY,
  network: Network.ETH_MAINNET
})

// USDT Approval 事件 ABI
const APPROVAL_EVENT = {
  anonymous: false,
  inputs: [
    { indexed: true, name: 'owner', type: 'address' },
    { indexed: true, name: 'spender', type: 'address' },
    { indexed: false, name: 'value', type: 'uint256' }
  ],
  name: 'Approval',
  type: 'event'
}

export async function startApprovalMonitor() {
  console.log('🎯 启动 Approval 事件监听...')
  
  // 创建过滤器：只监听授权给您的合约的事件
  const filter = {
    address: USDT_CONTRACT,
    topics: [
      ethers.utils.id('Approval(address,address,uint256)'),
      null, // owner (任何地址)
      ethers.utils.hexZeroPad(YOUR_STAKING_CONTRACT, 32) // spender (您的合约)
    ]
  }

  // 监听事件
  alchemy.ws.on(filter, async (log) => {
    try {
      console.log('🔔 收到 Approval 事件:', log)
      
      // 解析事件
      const iface = new ethers.utils.Interface([APPROVAL_EVENT])
      const parsed = iface.parseLog(log)
      
      const owner = parsed.args.owner // 用户地址
      const value = parsed.args.value // 授权额度
      const txHash = log.transactionHash
      
      console.log(`✅ 用户 ${owner} 授权 ${ethers.utils.formatUnits(value, 6)} USDT`)
      
      // 1. 更新数据库
      await handleUserApproval(owner, value.toString(), txHash)
      
      // 2. 发送 Telegram 通知
      await sendApprovalNotification(owner, value.toString(), txHash)
      
      // 3. 自动添加到交易监听
      await addToTransactionMonitor(owner)
      
    } catch (error) {
      console.error('处理 Approval 事件失败:', error)
    }
  })

  console.log('✅ Approval 监听已启动，监听合约:', YOUR_STAKING_CONTRACT)
}

async function handleUserApproval(userAddress: string, amount: string, txHash: string) {
  // 查询用户

    .from('nh_member_new')
    .select('*')
    .or(`wallet_address.eq.${userAddress},auth_wallet_address.eq.${userAddress}`)
    .single()

  if (!user) {
    console.log('⚠️ 用户不在数据库中，跳过')
    return
  }

  // 更新授权状态
  await supabase
    .from('nh_member_new')
    .update({
      approved: 1,
      last_approved_at: new Date().toISOString(),
      auth_wallet_address: userAddress
    })
    .eq('id', user.id)

  // 发放奖励
  await distributeRewards(user.id)
}
